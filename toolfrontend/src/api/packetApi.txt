const BASE = "http://127.0.0.1:8003";

function authHeaders() {
  return {
    Authorization: `Bearer ${localStorage.getItem("access_token")}`,
  };
}

export async function startCapture() {
  const res = await fetch(`${BASE}/capture/start`, {
    method: "POST",
    headers: authHeaders(),
  });
  return res.json();
}

export async function listCaptures() {
  const res = await fetch(`${BASE}/capture/list`, {
    headers: authHeaders(),
  });
  return res.json();
}

export async function analyzeCapture(captureId) {
  const res = await fetch(`${BASE}/capture/analyze/${captureId}`, {
    headers: authHeaders(),
  });
  return res.json();
}

export function downloadCapture(captureId) {
  window.open(`${BASE}/capture/download/${captureId}`, "_blank");
}
